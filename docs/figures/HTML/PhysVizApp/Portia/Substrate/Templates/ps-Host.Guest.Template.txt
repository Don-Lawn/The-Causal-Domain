// example extras for a %host% that can hold things, and can be indexed by fields of those things.
// {
//     "type": "Type:Thing.Host.%host%",
//     "name": "Name:myContainer",
//     "payload": {         
//         "field1": "value1",      
//         "field2": "value2",
//         "field3": "value3"
//     }
// }


//  this template is for a one to many relationship,  navigable ->


export default class %host% extends Thing{           // a bit like a Table in database.
    constructor(metadata) {
        super(metadata);
        this.%guest%Storage = new Array();  // the storage for the things in this %host%.  can be an array, a linked list, or something else.
        this.type = metadata.type ?? "Type:Thing.%host%";
        this.name = metadata.name;
        this.payload = metadata.payload;
        this.metadata = metadata;
        this.addAll%guest%(metadata)
    }

    get%guest%(slot) {        // return the %guest% at the specified slot in the %host%.
        return this.%guest%Storage[slot];
    }
    _set%guest%(slot, %guest%) {  // set the %guest% at the specified slot in the %host%.
        this.%guest%Storage[slot] = %guest%;
    }
    addAll%guest%(metadata) {
        for (const axisName of metadata.axes ?? [])
        {
            this.addNewAxis({
                name: axisName
            });
        }
    }

    addNew%guest%(extras) {  
        const new%guest% = new %guest% ("Thing.%guest%", extras.name, {},{  });
        const slot = this.%guest%Storage.push(new%guest%) - 1;
        for (const [fieldName, index] of this.indexes) {
            const value = this._getFieldValue(new%guest%, fieldName);
            index.addSlotEntry(value, slot);
        }
        return slot;
    }
    remove%guest%(slot) {  // remove the %guest% at the specified slot in the %host%.
        this.%guest%Storage.splice(slot, 1);
    }   


    addIndex (fieldName) {
        this.indexes.set(fieldName, new Index(fieldName,this));
    }


    findThing (fieldName, value) {
        // search for a thing in the %host% that matches the criteria, and return its location.
        const index= this.indexes.get(fieldName ?? "id")
        const slot = index.findSlotByKey(value);
        return this.%guest%Storage[slot];
    }

    _getFieldValue(obj, path) {
        let current = obj;

        for (const key of path.split('.')) {
            if (current == null) {
                return undefined;
            }
            current = current[key];
        }

        return current;
    }

    static selfTestCreate (metadata) {
        let testobj = new %host% (metadata.subRoles)
        console.log ("created new %host%", testObj)
    }
}
